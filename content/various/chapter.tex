\section{DP Optimizations}
    \subsection{Quadrangle Inequality (QI)}
        For $a \le b \le c \le d$, $C(a,c) + C(b,d) \le C(a,d) + C(b,c).$
    \subsection{Knuth Optimization}
        Applies to $dp[i][j] = cost[i][j] + \min_{i \le k < j} (dp[i][k] + dp[k+1][j]).$
        Requires $opt[i][j-1] \le opt[i][j] \le opt[i+1][j].$
        Sufficient: $cost$ satisfies QI and $cost[b][c] \le cost[a][d] \quad (a \le b \le c \le d).$
        Search $k \in [opt[i][j-1],\, opt[i+1][j]]$. Time $O(N^2)$.
    \subsection{Divide \& Conquer DP}
        $dp(i,j) = \min_{0 \le k \le j} (dp(i-1,k-1) + C(k,j)).$
        Requires $opt(i,j) \le opt(i,j+1)$. Sufficient: $C$ satisfies QI. Time $O(mn\log n)$.
    \subsection{1D1D Optimization}
        $dp[i] = \min_{j<i} (dp[j] + cost(j,i)).$
        Requires $par[i] \le par[i+1]$. Sufficient: $cost$ is QI. Time $O(n\log n)$.
